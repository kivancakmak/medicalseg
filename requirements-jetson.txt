# On-device INFERENCE requirements for the NVIDIA Jetson Nano 4GB.
# Target platform: JetPack 4.6 / L4T R32.6, Python 3.6.9, CUDA 10.2, TensorRT 8.x.
#
# This file is intentionally minimal. Install it inside a virtualenv created
# with --system-site-packages (see SETUP_JETSON.md) so that the system-provided
# packages below remain importable.
#
# DELIBERATELY NOT LISTED (do not add them):
#   * tensorrt  -> provided by the JetPack system package (python3-libnvinfer).
#                  There is no pip wheel for Jetson; `import tensorrt` resolves
#                  to /usr/lib/python3.6/dist-packages via --system-site-packages.
#   * cv2       -> OpenCV ships with JetPack (system package), built WITH
#                  GStreamer/CUDA support. The pip `opencv-python` wheel lacks
#                  GStreamer and would shadow the system build, breaking camera
#                  pipelines. Use the system cv2, do not pip install it.
#   * tensorflow / Pillow -> not needed for the TensorRT inference path on device;
#                  training and TFLite/Keras fallbacks stay on the desktop
#                  (see requirements-dev.txt).

# Pinned to match the numpy that JetPack 4.6 system packages were built against.
# Going higher risks ABI mismatches with the system cv2/scipy. 1.19.4 is the
# last reliable build for Python 3.6 on this platform.
numpy==1.19.4

# Host<->device memory + CUDA context management for the TensorRT engine.
# Built from source against CUDA 10.2; requires the CUDA env vars from
# SETUP_JETSON.md to be exported before `pip install`.
pycuda==2020.1

# Used to inspect / validate the exported .onnx model before TensorRT conversion.
onnx==1.11.0

# PCA9685 I2C PWM backend (pca9685_controller.py). Only needed when the external
# PCA9685 light driver board is actually attached; without these (or without the
# I2C bus) the controller degrades to a print-only stub.
adafruit-blinka
adafruit-circuitpython-pca9685
